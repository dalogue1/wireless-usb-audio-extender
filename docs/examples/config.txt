# USB Audio Extender settings
# Edit, save, then eject the EXTENDER drive: the local board restarts and
# passes the remote's settings on over the radio. Delete this file to
# return to the defaults.

mic = 1               # 1-9: the PC sees "Extender Mic N" (both boards follow)
channel = 11          # Wi-Fi channel 1-11 (both boards follow); pick a quiet one
feedback = voice       # voice or off (announcements in the recording; for beeps, copy
                       # the beeps voice pack into the voice folder)
noise_filter = on       # on (12 dB), strong (20 dB) or off (at the remote)
noise_adapt = normal   # normal, slow or fast: how quickly the filter follows changes
tx_power = max          # max, or 2-20 (dBm): radio transmit power of both boards
cpu_mhz = 240           # 240 or 160: processor speed of both boards (160 runs a little cooler)
highpass = on           # on, off: remove rumble, wind and hum below highpass_hz (remote)
highpass_hz = 200        # 50-1000: high-pass cut-off in Hz (200 keeps low owl and dove calls)
mic_level = default     # default, or 0-100: the adapter's microphone level, % of its range
                       # (default leaves the adapter's own level; status.txt shows it in dB;
                       # in speaker mode it sets the adapter's speaker level)
distance = normal       # near, normal or far: far reaches furthest but copes worst with busy
                       # Wi-Fi; near is the most robust where Wi-Fi is busy (both boards)
mode = auto             # auto, microphone or speaker: which way the audio goes. auto decides
                       # when the remote starts: a mic signal = microphone, silence = speaker
                       # (with a microphone always connected, microphone is the safe choice)
remote = any            # any, or the remote's MAC (status.txt): link only to that remote, and
                       # lock it to this local board. Back to any unlocks it again.
